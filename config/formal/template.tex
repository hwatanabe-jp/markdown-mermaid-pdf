  {\formalchapterbox{\thesection\hspace{1em}#1}}

  {\sffamily\bfseries\fontsize{12pt}{18pt}\selectfont\filcenter}

  {#1}

  {\sffamily\bfseries\fontsize{13pt}{19pt}\selectfont\color{formalheading}}

  {\sffamily\bfseries\fontsize{13pt}{19pt}\selectfont\color{formalheading}}

  {#1}
