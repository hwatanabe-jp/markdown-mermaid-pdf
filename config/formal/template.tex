\newcommand{\formalchapterbox}[1]{%
  \noindent\vbox{\offinterlineskip
    {\setlength{\fboxsep}{4pt}%
     \hbox{\colorbox{formalchapterband}{%
       \parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
         \strut\textcolor{formalheading}{#1}}}}}%

    \hbox{\textcolor{formalheading}{\rule{\linewidth}{0.8pt}}}}}
